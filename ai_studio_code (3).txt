class MockPaper {
  final String title;
  final String year;
  final String body; // UNEB, WAKISSHA, KASSHA, etc.
  final List<ExamQuestion> questions;

  MockPaper({
    required this.title,
    required this.year,
    required this.body,
    required this.questions,
  });
}

class ExamQuestion {
  final int number;
  final String text;
  final int marks;
  final String markingGuide;

  ExamQuestion({
    required this.number,
    required this.text,
    required this.marks,
    required this.markingGuide,
  });
}

class Subject {
  final String name;
  final String code;
  final IconData icon;
  final List<MockPaper> papers;

  Subject({
    required this.name,
    required this.code,
    required this.icon,
    required this.papers,
  });
}

class ExamUpdate {
  final String board; // UNEB, WAKISSHA, etc.
  final String title;
  final String date;
  final String summary;
  final bool isUrgent;

  ExamUpdate({
    required this.board,
    required this.title,
    required this.date,
    required this.summary,
    this.isUrgent = false,
  });
}

class ChatMessage {
  final String text;
  final bool isUser;
  final String time;

  ChatMessage({required this.text, required this.isUser, required this.time});
}

class SubjectGridListView extends StatelessWidget {
  final List<Subject> subjects;
  final String levelName;

  const SubjectGridListView({super.key, required this.subjects, required this.levelName});

  @override
  Widget build(BuildContext context) {
    return ListView.builder(
      padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 12),
      itemCount: subjects.length,
      itemBuilder: (context, index) {
        final sub = subjects[index];
        final paperCount = sub.papers.length;

        return Card(
          elevation: 1,
          margin: const EdgeInsets.symmetric(vertical: 6),
          shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
          child: ListTile(
            leading: CircleAvatar(
              backgroundColor: const Color(0xFF1E3A8A).withAlpha(25),
              child: Icon(sub.icon, color: const Color(0xFF1E3A8A)),
            ),
            title: Text(sub.name, style: const TextStyle(fontWeight: FontWeight.bold)),
            subtitle: Text('${sub.code} • ${paperCount == 0 ? "Papers in compilation" : "$paperCount Mock Paper Set(s)"}'),
            trailing: const Icon(Icons.arrow_forward_ios, size: 15, color: Colors.grey),
            onTap: () {
              Navigator.push(
                context,
                MaterialPageRoute(
                  builder: (context) => SubjectMocksScreen(subject: sub, levelName: levelName),
                ),
              );
            },
          ),
        );
      },
    );
  }
}

class PaperViewerScreen extends StatelessWidget {

  @override

    return Scaffold(
      appBar: AppBar(
        title: Text(paper.title),
        actions: [
          Padding(
            padding: const EdgeInsets.only(right: 12.0),
            child: Center(
              child: Chip(

                backgroundColor: const Color(0xFFD97706),
              ),
            ),
          )
        ],
      ),
      body: ListView.builder(
        padding: const EdgeInsets.all(14),
        itemCount: paper.questions.length,
        itemBuilder: (context, index) {
          final q = paper.questions[index];
          return QuestionGuideCard(question: q);
        },
      ),
    );
  }
}
