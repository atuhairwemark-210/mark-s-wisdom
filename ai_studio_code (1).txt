// ---------------- DATA MODELS ----------------
class MockPaper {
  final String title;
  final String year;
  final String body; // e.g., Wakiso Joint, Kampala Teachers, Mukono Mock
  final List<ExamQuestion> questions;

  MockPaper({
    required this.title,
    required this.year,
    required this.body,
    required this.questions,
  });
}

class ExamQuestion {
  final int questionNumber;
  final String questionText;
  final int marks;
  final String markingGuide; // Step-by-step points/rubric

  ExamQuestion({
    required this.questionNumber,
    required this.questionText,
    required this.marks,
    required this.markingGuide,
  });
}

class Subject {
  final String name;
  final IconData icon;
  final String code;
  final List<MockPaper> mockPapers;

  Subject({
    required this.name,
    required this.icon,
    required this.code,
    required this.mockPapers,
  });
}

// ---------------- COMPONENT: SUBJECT LIST ----------------
class SubjectListView extends StatelessWidget {
  final List<Subject> subjects;
  final String levelName;

  const SubjectListView({super.key, required this.subjects, required this.levelName});

  @override
  Widget build(BuildContext context) {
    return ListView.builder(
      padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 12),
      itemCount: subjects.length,
      itemBuilder: (context, index) {
        final subject = subjects[index];
        final paperCount = subject.mockPapers.length;

        return Card(
          elevation: 1.5,
          margin: const EdgeInsets.symmetric(vertical: 6),
          shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
          child: ListTile(
            leading: CircleAvatar(
              backgroundColor: Theme.of(context).colorScheme.primaryContainer,
              child: Icon(subject.icon, color: Theme.of(context).colorScheme.primary),
            ),
            title: Text(subject.name, style: const TextStyle(fontWeight: FontWeight.bold)),
            subtitle: Text('${subject.code} • ${paperCount == 0 ? "Papers coming soon" : "$paperCount Mock Paper(s)"}'),
            trailing: const Icon(Icons.arrow_forward_ios, size: 16),
            onTap: () {
              Navigator.push(
                context,
                MaterialPageRoute(
                  builder: (context) => SubjectMocksScreen(subject: subject, levelName: levelName),
                ),
              );
            },
          ),
        );
      },
    );
  }
}

// ---------------- SCREEN 2: MOCKS LIST FOR A SUBJECT ----------------
class SubjectMocksScreen extends StatelessWidget {
  final Subject subject;
  final String levelName;

  const SubjectMocksScreen({super.key, required this.subject, required this.levelName});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('${subject.name} ($levelName)'),
      ),
      body: subject.mockPapers.isEmpty
          ? Center(
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Icon(Icons.folder_open, size: 60, color: Colors.grey[400]),
                  const SizedBox(height: 12),
                  Text('No Mock Papers added for ${subject.name} yet.\nCheck back soon!',
                      textAlign: TextAlign.center,
                      style: const TextStyle(fontSize: 16, color: Colors.grey)),
                ],
              ),
            )
          : ListView.builder(
              padding: const EdgeInsets.all(14),
              itemCount: subject.mockPapers.length,
              itemBuilder: (context, index) {
                final mock = subject.mockPapers[index];
                return Card(
                  margin: const EdgeInsets.only(bottom: 12),
                  child: ListTile(
                    contentPadding: const EdgeInsets.all(12),
                    title: Text(mock.title, style: const TextStyle(fontWeight: FontWeight.bold)),
                    subtitle: Column(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        const SizedBox(height: 4),
                        Text('Agency: ${mock.body}'),
                        Text('Year: ${mock.year} • ${mock.questions.length} Question(s) with Guides'),
                      ],
                    ),
                    trailing: ElevatedButton(
                      onPressed: () {
                        Navigator.push(
                          context,
                          MaterialPageRoute(
                            builder: (context) => MockPaperViewerScreen(mockPaper: mock),
                          ),
                        );
                      },
                      child: const Text('Open'),
                    ),
                  ),
                );
              },
            ),
    );
  }
}

// ---------------- SCREEN 3: MOCK PAPER & MARKING GUIDE VIEWER ----------------
class MockPaperViewerScreen extends StatelessWidget {
  final MockPaper mockPaper;
  const MockPaperViewerScreen({super.key, required this.mockPaper});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(mockPaper.title),
        actions: [
          Padding(
            padding: const EdgeInsets.only(right: 12),
            child: Chip(
              label: Text(mockPaper.year, style: const TextStyle(color: Colors.white, fontWeight: FontWeight.bold)),
              backgroundColor: Theme.of(context).colorScheme.primary,
            ),
          )
        ],
      ),
      body: mockPaper.questions.isEmpty
          ? const Center(child: Text('Questions uploading...'))
          : ListView.builder(
              padding: const EdgeInsets.all(16),
              itemCount: mockPaper.questions.length,
              itemBuilder: (context, index) {
                final q = mockPaper.questions[index];
                return QuestionWithGuideCard(question: q);
              },
            ),
    );
  }
}

// ---------------- WIDGET: QUESTION + TOGGLEABLE MARKING GUIDE ----------------
class QuestionWithGuideCard extends StatefulWidget {
  final ExamQuestion question;
  const QuestionWithGuideCard({super.key, required this.question});

  @override
  State<QuestionWithGuideCard> createState() => _QuestionWithGuideCardState();
}

class _QuestionWithGuideCardState extends State<QuestionWithGuideCard> {
  bool _showGuide = false;

  @override
  Widget build(BuildContext context) {
    return Card(
      elevation: 2,
      margin: const EdgeInsets.only(bottom: 16),
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
      child: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Row(
              mainAxisAlignment: MainAxisAlignment.spaceBetween,
              children: [
                Text(
                  'Question ${widget.question.questionNumber}',
                  style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 16),
                ),
                Container(
                  padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 3),
                  decoration: BoxDecoration(
                    color: Colors.grey[200],
                    borderRadius: BorderRadius.circular(6),
                  ),
                  child: Text(
                    '[${widget.question.marks} Marks]',
                    style: const TextStyle(fontWeight: FontWeight.bold),
                  ),
                ),
              ],
            ),
            const Divider(height: 20),
            Text(
              widget.question.questionText,
              style: const TextStyle(fontSize: 16, height: 1.4),
            ),
            const SizedBox(height: 16),

            // Toggle Marking Guide Button
            OutlinedButton.icon(
              style: OutlinedButton.styleFrom(
                foregroundColor: _showGuide ? Colors.red : Colors.green[800],
                side: BorderSide(color: _showGuide ? Colors.red : Colors.green),
              ),
              onPressed: () {
                setState(() {
                  _showGuide = !_showGuide;
                });
              },
              icon: Icon(_showGuide ? Icons.visibility_off : Icons.check_circle_outline),
              label: Text(_showGuide ? 'Hide Marking Guide' : 'Reveal Marking Guide & Answers'),
            ),

            // The Marking Guide section
            if (_showGuide) ...[
              const SizedBox(height: 12),
              Container(
                width: double.infinity,
                padding: const EdgeInsets.all(14),
                decoration: BoxDecoration(
                  color: Colors.green.withAlpha(20),
                  borderRadius: BorderRadius.circular(8),
                  border: Border.all(color: Colors.green.shade300),
                ),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    const Row(
                      children: [
                        Icon(Icons.rule, size: 20, color: Colors.green),
                        SizedBox(width: 8),
                        Text(
                          'Official Marking Guide / Scoring Rubric',
                          style: TextStyle(fontWeight: FontWeight.bold, color: Colors.green),
                        ),
                      ],
                    ),
                    const SizedBox(height: 8),
                    Text(
                      widget.question.markingGuide,
                      style: const TextStyle(fontSize: 14.5, height: 1.5, fontFamily: 'monospace'),
                    ),
                  ],
                ),
              ),
            ],
          ],
        ),
      ),
    );
  }
}
